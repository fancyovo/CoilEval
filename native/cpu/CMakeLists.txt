cmake_minimum_required(VERSION 3.22)
project(coil_eval_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(OpenMP)
# The configuration and host selection policy have one maintained source.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../src/pipeline/api.cu" api_source)
string(FIND "${api_source}" "int sgpu_default_score_config(" defaults_begin)
string(FIND "${api_source}" "int sgpu_score_coils(" defaults_end)
math(EXPR defaults_length "${defaults_end}-${defaults_begin}")
string(SUBSTRING "${api_source}" ${defaults_begin} ${defaults_length} defaults_source)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/defaults.inc" "${defaults_source}")
add_library(coil_eval_cpu SHARED field.cpp simd_avx2.cpp autograd.cpp autograd_simd_avx2.cpp geometry.cpp axis_host.cpp axis_polish.cpp common_host.cpp coil_geometry_host.cpp config.cpp)
target_include_directories(coil_eval_cpu PRIVATE include ../include "${CMAKE_CURRENT_BINARY_DIR}")
if(OpenMP_CXX_FOUND)
  target_link_libraries(coil_eval_cpu PRIVATE OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_options(coil_eval_cpu PRIVATE /O2)
  set_target_properties(coil_eval_cpu PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
else()
  target_compile_options(coil_eval_cpu PRIVATE -O3)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_link_options(coil_eval_cpu PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endif()
install(TARGETS coil_eval_cpu LIBRARY DESTINATION coil_eval/_native RUNTIME DESTINATION coil_eval/_native)
